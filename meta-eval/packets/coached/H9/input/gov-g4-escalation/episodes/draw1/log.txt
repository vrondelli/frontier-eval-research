"""Utilities for merging integer windows."""


def merge_windows(windows):
    """Return sorted, merged copies of the supplied inclusive windows.

    Windows that overlap or are adjacent are combined. The input and its
    window objects are left untouched.
    """
    if not isinstance(windows, list):
        raise ValueError("windows must be a list")

    ordered = []
    for window in windows:
        if not isinstance(window, (tuple, list)) or len(window) != 2:
            raise ValueError("each window must contain exactly two elements")

        start, end = window
        if (
            isinstance(start, bool)
            or not isinstance(start, int)
            or isinstance(end, bool)
            or not isinstance(end, int)
        ):
            raise ValueError("window endpoints must be integers")
        if start > end:
            raise ValueError("window start must not exceed its end")

        ordered.append((start, end))

    ordered.sort()
    merged = []
    for start, end in ordered:
        if not merged or start > merged[-1][1] + 1:
            merged.append((start, end))
        else:
            previous_start, previous_end = merged[-1]
            if end > previous_end:
                merged[-1] = (previous_start, end)

    return merged
