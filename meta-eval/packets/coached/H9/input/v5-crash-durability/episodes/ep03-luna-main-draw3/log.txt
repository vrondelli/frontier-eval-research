    def test_append_and_read_back(self):
        s1 = ledger.append_record(self.path, "alice", 50, "first")

                {"seq": 2, "account": "bob", "amount": -20, "note": ""},
                {"seq": 3, "account": "alice", "amount": 5, "note": "third"},

        self.assertEqual(ledger.balance(self.path, "alice"), 55)
        self.assertEqual(ledger.balance(self.path, "bob"), -20)

"""A small process-safe, crash-durable file-backed ledger."""

import fcntl
import json
import os
import tempfile
from contextlib import contextmanager


def _store_paths(path):
    """Return normalized store and lock paths for a path-like argument."""
    store = os.path.abspath(os.fsdecode(os.fspath(path)))
    return store, store + ".lock"


@contextmanager
def _locked(path, exclusive):
    store, lock_path = _store_paths(path)
    directory = os.path.dirname(store)
    # Creating the parent also makes first use safe when the directory itself
    # has not been created yet. os.makedirs(exist_ok=True) is race-safe.
    os.makedirs(directory, exist_ok=True)
    fd = os.open(lock_path, os.O_CREAT | os.O_RDWR, 0o666)
    try:
        fcntl.flock(fd, fcntl.LOCK_EX if exclusive else fcntl.LOCK_SH)
        yield store, directory
    finally:
        try:
            fcntl.flock(fd, fcntl.LOCK_UN)
        finally:
            os.close(fd)


def _read_records(store):
    try:
        with open(store, "r", encoding="ascii") as stream:
            records = json.load(stream)
    except FileNotFoundError:
        return []

    # The file is private implementation state, but reject unexpected content
    # instead of silently returning misleading ledger values.
    if not isinstance(records, list):
        raise ValueError("invalid ledger store")
    return records


def _write_records(store, directory, records):
    # ASCII JSON safely represents every Python string, including strings with
    # lone surrogate code points. The complete new image is written before it
    # can replace the previous durable image.
    payload = json.dumps(records, ensure_ascii=True, separators=(",", ":"))
    data = payload.encode("ascii")
    fd, temporary = tempfile.mkstemp(prefix=".ledger-", suffix=".tmp", dir=directory)
    try:
        with os.fdopen(fd, "wb") as stream:
            stream.write(data)
            stream.flush()
            os.fsync(stream.fileno())
        os.replace(temporary, store)
        # Persist the directory entry update before reporting success.
        dir_fd = os.open(directory, os.O_RDONLY | getattr(os, "O_DIRECTORY", 0))
        try:
            os.fsync(dir_fd)
        finally:
            os.close(dir_fd)
    finally:
        try:
            os.unlink(temporary)
        except FileNotFoundError:
            pass


def append_record(path, account, amount, note=""):
    if not isinstance(account, str) or account == "":
        raise ValueError("account must be a non-empty string")
    if not isinstance(amount, int) or isinstance(amount, bool):
        raise ValueError("amount must be an int and not bool")
    if not isinstance(note, str):
        raise ValueError("note must be a string")

    with _locked(path, exclusive=True) as (store, directory):
        records = _read_records(store)
        seq = len(records) + 1
        records.append(
            {"seq": seq, "account": account, "amount": amount, "note": note}
        )
        _write_records(store, directory, records)
        return seq


def list_records(path):
    with _locked(path, exclusive=False) as (store, _directory):
        # Copy into the documented shape so callers cannot mutate internal
        # decoded state and every result has exactly the public fields.
        return [
            {
                "seq": record["seq"],
                "account": record["account"],
                "amount": record["amount"],
                "note": record["note"],
            }
            for record in _read_records(store)
        ]


def balance(path, account):
    with _locked(path, exclusive=False) as (store, _directory):
        return sum(
            record["amount"]
            for record in _read_records(store)
            if record["account"] == account
        )


def count(path):
    with _locked(path, exclusive=False) as (store, _directory):
        return len(_read_records(store))
