"""Exact clinic billing settlement."""

import re
from decimal import Decimal, ROUND_CEILING, ROUND_HALF_UP, localcontext


_CODE = re.compile(r"(?:[0-9]{7}|[0-9]{7}-[0-9]{3})\Z", re.ASCII)
_CHARGE = re.compile(r"[0-9]+(?:\.[0-9]+)?\Z", re.ASCII)
_WAIVER = re.compile(r"-?[0-9]+(?:\.[0-9]+)?\Z", re.ASCII)
_DATE = re.compile(r"([0-9]{4})-([0-9]{2})-([0-9]{2})\Z", re.ASCII)

_BANDS = (
    (31, "basic", Decimal("3"), Decimal("0.11")),
    (64, "standard", Decimal("5"), Decimal("0.17")),
    (88, "enhanced", Decimal("8"), Decimal("0.24")),
    (99, "intensive", Decimal("13"), Decimal("0.31")),
)


def _band_for(prefix):
    for upper, band, _levy, _rate in _BANDS:
        if prefix <= upper:
            return band
    return "intensive"


def _round_mean(total, count):
    """Round an integer ratio to nearest integer, ties away from zero."""
    sign = -1 if total < 0 else 1
    quotient, remainder = divmod(abs(total), count)
    if remainder * 2 >= count:
        quotient += 1
    return sign * quotient


def settle_bills(records):
    """Settle eligible records using exact decimal arithmetic."""
    excluded = []
    bills = {}
    per_band = {}
    total_minor = 0

    for record in records:
        patient = record["patient"]
        code = record["account"]
        consult_text = record["consult"]
        lab_text = record["lab"]
        date_text = record["admitted"]
        waiver_text = record["waiver"]

        # Validate all fields before doing any arithmetic so each rejected
        # record is reported exactly once.
        if not _CODE.fullmatch(code):
            excluded.append(patient)
            continue
        if not _CHARGE.fullmatch(consult_text) or not _CHARGE.fullmatch(lab_text):
            excluded.append(patient)
            continue
        if not _WAIVER.fullmatch(waiver_text):
            excluded.append(patient)
            continue
        date_match = _DATE.fullmatch(date_text)
        if not date_match:
            excluded.append(patient)
            continue

        year, month, day = map(int, date_match.groups())
        if not (1 <= month <= 12 and 1 <= day <= 31):
            excluded.append(patient)
            continue
        if (year, month, day) > (2025, 3, 20):
            excluded.append(patient)
            continue

        prefix = int(code[:2])
        band = _band_for(prefix)
        # Use precision proportional to the input size. Decimal's default
        # context can otherwise round valid, very long input strings.
        precision = len(consult_text) + len(lab_text) + len(waiver_text) + 50
        with localcontext() as ctx:
            ctx.prec = max(precision, 50)
            consult = Decimal(consult_text)
            lab = Decimal(lab_text)
            waiver = Decimal(waiver_text)
            combined = consult + lab
            discount = min(combined * Decimal("0.09"), Decimal("18"))
            raw_billable = combined - discount
            billable = (raw_billable * 4).to_integral_value(rounding=ROUND_CEILING) / 4

            for _upper, candidate, levy, rate in _BANDS:
                if candidate == band:
                    break
            subtotal = levy + rate * billable + Decimal("2")
            amount = subtotal * Decimal("1.07")
            if band == "intensive":
                amount += Decimal("6")
            amount -= waiver
            bill = int((amount * 100).quantize(Decimal("1"), rounding=ROUND_HALF_UP))

        bills[patient] = bill
        per_band[band] = per_band.get(band, 0) + bill
        total_minor += bill

    count = len(bills)
    return {
        "total_minor": total_minor,
        "billed": count,
        "excluded": excluded,
        "bills": bills,
        "per_band": per_band,
        "mean_minor": _round_mean(total_minor, count) if count else 0,
    }
