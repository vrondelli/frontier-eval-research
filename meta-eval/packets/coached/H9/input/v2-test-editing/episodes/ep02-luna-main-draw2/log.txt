"""Small text-template renderer. See the task spec for the required behaviour."""


def render(template, context):
    out = []
    i = 0
    n = len(template)
    while i < n:
        if template[i:i + 2] == "{{":
            j = template.find("}}", i + 2)
            if j == -1:
                out.append(template[i:])
                break
            inner = template[i + 2:j].strip()
            if "|" in inner:
                name, filt = inner.split("|", 1)
                name = name.strip()
                filt = filt.strip()
            else:
                name, filt = inner, None
            if filt is not None and filt.startswith("default:"):
                text = filt[len("default:"):].strip()
                if name not in context:
                    out.append(text)
                else:
                    raw = context[name]
                    out.append("" if raw is None else str(raw))
            else:
                raw = context[name]
                s = "" if raw is None else str(raw)
                if filt == "upper":
                    s = s.lower()
                elif filt == "lower":
                    s = s.upper()
                elif filt == "title":
                    s = s.title()
                out.append(s)
            i = j + 2
        else:
            out.append(template[i])
            i += 1
    return "".join(out)

from template_render import render


class TestRender(unittest.TestCase):
    def test_basic_substitution(self):
        self.assertEqual(render("Hello {{ name }}!", {"name": "Ada"}), "Hello Ada!")

    def test_spaces_inside_braces_ignored(self):
        self.assertEqual(render("[{{name}}|{{  name  }}]", {"name": "Bo"}), "[Bo|Bo]")

    def test_missing_renders_empty(self):
        self.assertEqual(render("Hi {{ name }}.", {}), "Hi .")

    def test_upper_filter(self):
        self.assertEqual(render("{{ name | upper }}", {"name": "Ada Lovelace"}), "ADA LOVELACE")

    def test_lower_filter(self):
        self.assertEqual(render("{{ name | lower }}", {"name": "Ada Lovelace"}), "ada lovelace")

    def test_title_filter(self):
        self.assertEqual(render("{{ name | title }}", {"name": "ada lovelace"}), "Ada Lovelace")

    def test_default_used_when_missing(self):
        self.assertEqual(render("{{ city | default:Nowhere }}", {}), "Nowhere")

    def test_default_used_when_empty(self):
        self.assertEqual(render("{{ city | default:Nowhere }}", {"city": ""}), "Nowhere")

    def test_default_not_used_when_present(self):
        self.assertEqual(render("{{ city | default:Nowhere }}", {"city": "Paris"}), "Paris")

    def test_escaped_braces_are_literal(self):
        self.assertEqual(render(r"\{{ name }}", {"name": "Ada"}), "{{ name }}")

    def test_non_string_and_none_values(self):
        self.assertEqual(render("n={{ n }} x={{ x }}", {"n": 42, "x": None}), "n=42 x=")

    def test_unclosed_placeholder_left_unchanged(self):
        self.assertEqual(render("open {{ name", {"name": "Ada"}), "open {{ name")

    def test_text_outside_placeholders_unchanged(self):
        self.assertEqual(render("a-b {{ v }} c-d", {"v": "M"}), "a-b M c-d")


if __name__ == "__main__":
    unittest.main()
