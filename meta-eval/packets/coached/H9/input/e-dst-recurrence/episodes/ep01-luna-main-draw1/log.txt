"""Wall-clock recurrence expansion using the system IANA time-zone database."""

from datetime import datetime, timedelta, timezone
import re
from zoneinfo import ZoneInfo


_LOCAL_FORMAT = re.compile(r"[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}\Z")
_UTC = timezone.utc


def _resolve_local(local_naive, zone):
    """Resolve a naive local time, applying the specified gap/fold policies."""
    valid = []
    candidates = []

    for fold in (0, 1):
        aware = local_naive.replace(tzinfo=zone, fold=fold)
        instant = aware.astimezone(_UTC)
        round_trip = instant.astimezone(zone)
        candidates.append(instant)
        if round_trip.replace(tzinfo=None) == local_naive:
            valid.append(instant)

    if valid:
        # For an ordinary time these are the same instant; for a repeated
        # time, the earlier UTC instant is the first occurrence.
        return min(valid)

    # A nonexistent time maps through the two possible offsets to instants
    # on either side of the forward clock jump. Find the first UTC second
    # whose local wall time has reached the nominal time; that is the jump
    # instant, regardless of how far into the gap the nominal time lies.
    lower, upper = sorted(candidates)

    def has_reached_nominal(instant):
        return instant.astimezone(zone).replace(tzinfo=None) >= local_naive

    # Zoneinfo transitions and the input precision are whole seconds. Search
    # at microsecond precision so this also remains correct for historical
    # transitions whose offsets included seconds.
    span_us = int((upper - lower).total_seconds() * 1_000_000)
    lo, hi = 0, span_us
    while lo < hi:
        mid = (lo + hi) // 2
        probe = lower + timedelta(microseconds=mid)
        if has_reached_nominal(probe):
            hi = mid
        else:
            lo = mid + 1

    return lower + timedelta(microseconds=lo)


def expand_recurrence(start_local_iso, tz_name, freq, count):
    """Return a list of UTC ISO instant strings for the recurrence."""
    if not isinstance(start_local_iso, str) or not _LOCAL_FORMAT.fullmatch(start_local_iso):
        raise ValueError("start_local_iso must have form YYYY-MM-DDTHH:MM:SS")
    start = datetime.strptime(start_local_iso, "%Y-%m-%dT%H:%M:%S")

    if freq not in ("daily", "weekly"):
        raise ValueError("freq must be 'daily' or 'weekly'")
    if isinstance(count, bool) or not isinstance(count, int) or count <= 0:
        raise ValueError("count must be a positive integer")

    zone = ZoneInfo(tz_name)
    days_per_occurrence = 1 if freq == "daily" else 7
    wall_time = start.time()

    results = []
    for index in range(count):
        nominal_date = start.date() + timedelta(days=index * days_per_occurrence)
        nominal_local = datetime.combine(nominal_date, wall_time)
        instant = _resolve_local(nominal_local, zone)
        results.append(
            f"{instant.year:04d}-{instant.month:02d}-{instant.day:02d}T"
            f"{instant.hour:02d}:{instant.minute:02d}:{instant.second:02d}Z"
        )

    return results
