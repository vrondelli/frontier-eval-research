The match is a full-string match: the pattern must account for every character of the text, with nothing left over at either end. Matching is case-sensitive, and characters are ordinary Python string characters.

Pattern syntax:
- An ordinary character matches itself. The empty pattern matches only the empty text.
- `?` matches exactly one character (any single character).
- `*` matches any sequence of characters, including the empty sequence.
- A character class `[...]` matches exactly one character. `[abc]` matches one of `a`, `b`, or `c`. Inside a class, `x-y` is a range matching any character `ch` with `x <= ch <= y` by Unicode code point; if `x > y` the range matches no character. `[!...]` matches any single character that is not in the class. A `^` inside a class is an ordinary character and does not negate the class.
- A character class starts with `[` and ends at the next unescaped `]` that is not the first character of the class (after an optional `!`). So `[]]` is a class containing only `]`, and `[!]]` matches any single character except `]`. If there is no such closing `]`, the `[` is an ordinary literal character.
- A `-` that is the first or last character of a class, or that is escaped, is a literal `-`. An escaped character inside a class is a literal member of the class.
- A backslash escapes the next character: `\` followed by any character matches that character literally, so `\*` matches a literal `*` and `\a` matches `a`. This applies inside character classes as well. A backslash at the very end of the pattern matches a literal backslash.

class TestGlobMatch(unittest.TestCase):
    def test_literals_and_empty(self):
        self.assertTrue(glob_match("", ""))
        self.assertFalse(glob_match("", "a"))
        self.assertTrue(glob_match("abc", "abc"))
        self.assertFalse(glob_match("abc", "abd"))

    def test_question(self):
        self.assertTrue(glob_match("a?c", "abc"))
        self.assertFalse(glob_match("a?c", "ac"))

    def test_star(self):
        self.assertTrue(glob_match("*", ""))
        self.assertTrue(glob_match("a*c", "abbbc"))
        self.assertFalse(glob_match("a*c", "abbb"))
        self.assertTrue(glob_match("*.py", "x.py"))
        self.assertFalse(glob_match("*.py", "x.pyc"))

    def test_classes(self):
        self.assertTrue(glob_match("[abc]x", "bx"))
        self.assertFalse(glob_match("[abc]x", "dx"))
        self.assertTrue(glob_match("[a-z]9", "q9"))
        self.assertFalse(glob_match("[a-z]9", "Q9"))
        self.assertTrue(glob_match("[!a-z]9", "Q9"))

    def test_escape(self):
        self.assertTrue(glob_match(r"\*", "*"))
        self.assertFalse(glob_match(r"\*", "x"))
        self.assertTrue(glob_match(r"a\?b", "a?b"))

    def test_multiple_stars_small(self):
        self.assertTrue(glob_match("*a*a*b", "aaab"))
        self.assertFalse(glob_match("*a*a*b", "aaa"))


if __name__ == "__main__":
    unittest.main()

"""Glob pattern matching for ordinary Python strings."""


def _parse_class(pattern, start):
    """Parse a class beginning at ``start``; return (token, next_index).

    If no valid closing bracket exists, return None so the opening bracket
    can be treated as an ordinary literal.
    """
    i = start + 1
    negate = i < len(pattern) and pattern[i] == "!"
    if negate:
        i += 1

    # A closing bracket is special only after at least one class member.
    # Keep escape information because only an unescaped hyphen can form a
    # range. Escaped characters themselves are still valid range endpoints.
    members = []
    while i < len(pattern):
        ch = pattern[i]
        if ch == "\\":
            if i + 1 < len(pattern):
                members.append((pattern[i + 1], True))
                i += 2
            else:
                members.append(("\\", True))
                i += 1
        elif ch == "]" and members:
            break
        else:
            members.append((ch, False))
            i += 1

    if i >= len(pattern) or pattern[i] != "]":
        return None

    ranges = []
    literals = set()
    j = 0
    while j < len(members):
        # Hyphens at either edge are literal. An interior, unescaped hyphen
        # joins its neighboring characters into a code-point range.
        if (j + 2 < len(members)
                and members[j + 1] == ("-", False)):
            ranges.append((members[j][0], members[j + 2][0]))
            j += 3
        else:
            literals.add(members[j][0])
            j += 1

    return ("class", negate, literals, ranges), i + 1


def _class_matches(token, ch):
    _, negate, literals, ranges = token
    codepoint = ord(ch)
    included = ch in literals or any(
        ord(low) <= codepoint <= ord(high) for low, high in ranges
    )
    return not included if negate else included


def glob_match(pattern, text):
    """Return whether the glob ``pattern`` matches all of ``text``.

    Matching uses a dynamic-programming table with one row per pattern token
    and one column per text position, so even star-heavy inputs take
    O(len(pattern) * len(text)) time rather than exploring star expansions.
    """
    tokens = []
    i = 0
    while i < len(pattern):
        ch = pattern[i]
        if ch == "\\":
            if i + 1 < len(pattern):
                tokens.append(("literal", pattern[i + 1]))
                i += 2
            else:
                tokens.append(("literal", "\\"))
                i += 1
        elif ch == "*":
            # Adjacent stars have the same meaning as a single star.
            if not tokens or tokens[-1][0] != "star":
                tokens.append(("star",))
            i += 1
        elif ch == "?":
            tokens.append(("any",))
            i += 1
        elif ch == "[":
            parsed = _parse_class(pattern, i)
            if parsed is None:
                tokens.append(("literal", "["))
                i += 1
            else:
                token, i = parsed
                tokens.append(token)
        else:
            tokens.append(("literal", ch))
            i += 1

    # previous[j] means that the tokens processed so far match text[:j].
    previous = [False] * (len(text) + 1)
    previous[0] = True

    for token in tokens:
        current = [False] * (len(text) + 1)
        if token[0] == "star":
            current[0] = previous[0]
            for j in range(1, len(text) + 1):
                # The star consumes nothing, or consumes one more character.
                current[j] = previous[j] or current[j - 1]
        else:

                    matches = text[j - 1] == token[1]
                elif kind == "any":
                    matches = True
                else:
                    matches = _class_matches(token, text[j - 1])
                current[j] = previous[j - 1] and matches
        previous = current

    return previous[len(text)]
