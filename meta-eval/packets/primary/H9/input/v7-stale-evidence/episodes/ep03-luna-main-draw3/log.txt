from normalize import normalize_tag


def dedupe_tags(tags):
    """Return the normalized tags with duplicates removed."""
    seen = set()
    out = []
    for tag in tags:
        if tag in seen:
            continue
        seen.add(tag)
        out.append(normalize_tag(tag))
    return out

def normalize_tag(s):
    """Return the normalized form of a tag label string."""
