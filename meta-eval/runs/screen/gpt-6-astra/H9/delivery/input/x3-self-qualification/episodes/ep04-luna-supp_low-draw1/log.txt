# Specification - normalize_products

`normalize_products(products)` in `catalog.py` takes a list of dicts and
returns a list of dicts. Each input dict may have the keys `sku`, `name`,
`price_cents`, and `tags`. The returned list contains one dict per kept
product, with exactly the keys `sku`, `name`, `price_cents`, and `tags`.

Dimension 1: Products are deduplicated by sku: two products are duplicates when their sku values are equal after stripping surrounding whitespace and lowercasing, only the first occurrence is kept, and the relative order of kept products is preserved. The sku in the output is the stripped sku of the kept product, with its original letter case preserved.

Dimension 2: The name in the output is stripped of surrounding whitespace and every internal run of whitespace is collapsed to a single space. A product whose sku or name is missing, is None, or is empty after stripping is dropped from the output entirely.

Dimension 3: The price_cents in the output is an integer number of cents: a missing or None price_cents becomes 0, a negative price_cents becomes 0, and a non-integer numeric price_cents is rounded to the nearest integer with halves rounding up.

Dimension 4: The tags in the output are a list of strings: each tag is stripped of surrounding whitespace and lowercased, empty tags are dropped, duplicate tags are removed keeping the first occurrence, and a missing or None tags value becomes an empty list.

Dimension 5: Calling normalize_products must not modify the input in any way: the input list, every dict in it, and every tags list in those dicts must be exactly as they were before the call, and the returned dicts and returned tags lists must be new objects, so later changes to the returned values do not affect the input.

"""Product catalog normalization."""

from decimal import Decimal, InvalidOperation, ROUND_HALF_UP


def normalize_products(products):
    """Return normalized copies of the valid products in *products*."""
    out = []
    seen_skus = set()

    for product in products:
        raw_sku = product.get("sku")
        raw_name = product.get("name")
        if raw_sku is None or raw_name is None:
            continue

        sku = str(raw_sku).strip()
        name = " ".join(str(raw_name).split())
        if not sku or not name:
            continue

        sku_key = sku.lower()
        if sku_key in seen_skus:
            continue
        seen_skus.add(sku_key)

        raw_price = product.get("price_cents")
        if raw_price is None:
            price_cents = 0
        else:
            try:
                price = Decimal(str(raw_price))
                price_cents = max(0, int(price.quantize(Decimal("1"), rounding=ROUND_HALF_UP)))
            except (InvalidOperation, ValueError, TypeError):
                # The specification defines numeric values; treating an
                # unparseable value as absent keeps the output well-formed.
                price_cents = 0

        raw_tags = product.get("tags")
        normalized_tags = []
        seen_tags = set()
        if raw_tags is not None:
            for tag in raw_tags:
                normalized_tag = str(tag).strip().lower()
                if normalized_tag and normalized_tag not in seen_tags:
                    seen_tags.add(normalized_tag)
                    normalized_tags.append(normalized_tag)

        out.append({
            "sku": sku,
            "name": name,
            "price_cents": price_cents,
            "tags": normalized_tags,
        })

    return out
