import fcntl
import json
import os
import tempfile
from contextlib import contextmanager


def _paths(path):
    """Return a normalized ledger path and its stable sidecar lock path."""
    ledger_path = os.path.abspath(os.fsdecode(os.fspath(path)))
    return ledger_path, ledger_path + ".lock"


@contextmanager
def _locked(path, exclusive):
    ledger_path, lock_path = _paths(path)
    flags = os.O_CREAT | os.O_RDWR
    lock_fd = os.open(lock_path, flags, 0o666)
    try:
        fcntl.flock(lock_fd, fcntl.LOCK_EX if exclusive else fcntl.LOCK_SH)
        try:
            yield ledger_path
        finally:
            fcntl.flock(lock_fd, fcntl.LOCK_UN)
    finally:
        os.close(lock_fd)


def _read_records(ledger_path):
    try:
        with open(ledger_path, "r", encoding="utf-8") as stream:
            records = json.load(stream)
    except FileNotFoundError:
        return []
    if not isinstance(records, list):
        raise ValueError("ledger store is not a record list")
    return records


def append_record(path, account, amount, note=""):
    if not isinstance(account, str) or not account:
        raise ValueError("account must be a non-empty string")
    if type(amount) is not int:
        raise ValueError("amount must be an int and not a bool")
    if not isinstance(note, str):
        raise ValueError("note must be a string")

    with _locked(path, exclusive=True) as ledger_path:
        records = _read_records(ledger_path)
        seq = len(records) + 1
        records.append(
            {"seq": seq, "account": account, "amount": amount, "note": note}
        )

        directory = os.path.dirname(ledger_path) or "."
        fd, temporary_path = tempfile.mkstemp(prefix=".ledger-", dir=directory)

                json.dump(records, stream, ensure_ascii=False, separators=(",", ":"))
