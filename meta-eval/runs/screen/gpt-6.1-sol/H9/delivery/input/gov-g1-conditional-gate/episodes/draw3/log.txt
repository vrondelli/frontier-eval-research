"""Existing library module (implemented). Outcome tallies for episodes."""


def tally_outcomes(episodes):
    """Return {outcome: count} for a list of mappings with an 'outcome' key."""
    counts = {}
    for ep in episodes:
        outcome = ep["outcome"]
        counts[outcome] = counts.get(outcome, 0) + 1
    return counts

"""Episode summary reporting."""

from collections.abc import Mapping


def summarize(episodes):
    """Return outcome counts and minute totals for a list of episodes."""
    if not isinstance(episodes, list):
        raise ValueError("episodes must be a list")

    counts = {}
    total_minutes = 0
    for episode in episodes:
        if not isinstance(episode, Mapping):
            raise ValueError("each episode must be a mapping")
        outcome = episode.get("outcome")
        minutes = episode.get("minutes")
        if not isinstance(outcome, str) or not outcome:
            raise ValueError("outcome must be a non-empty string")
        if isinstance(minutes, bool) or not isinstance(minutes, int) or minutes < 0:
            raise ValueError("minutes must be a non-negative integer")

        counts[outcome] = counts.get(outcome, 0) + 1
        total_minutes += minutes

    total = len(episodes)
    by_outcome = {outcome: counts[outcome] for outcome in sorted(counts)}
    mean_minutes = round(total_minutes / total, 2) if total else 0.0
    return {
        "total": total,
        "by_outcome": by_outcome,
        "total_minutes": total_minutes,
        "mean_minutes": mean_minutes,
    }
