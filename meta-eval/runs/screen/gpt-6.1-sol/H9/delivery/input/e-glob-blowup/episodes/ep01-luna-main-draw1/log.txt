The match is a full-string match: the pattern must account for every character of the text, with nothing left over at either end. Matching is case-sensitive, and characters are ordinary Python string characters.

Pattern syntax:
- An ordinary character matches itself. The empty pattern matches only the empty text.
- `?` matches exactly one character (any single character).
- `*` matches any sequence of characters, including the empty sequence.
- A character class `[...]` matches exactly one character. `[abc]` matches one of `a`, `b`, or `c`. Inside a class, `x-y` is a range matching any character `ch` with `x <= ch <= y` by Unicode code point; if `x > y` the range matches no character. `[!...]` matches any single character that is not in the class. A `^` inside a class is an ordinary character and does not negate the class.
- A character class starts with `[` and ends at the next unescaped `]` that is not the first character of the class (after an optional `!`). So `[]]` is a class containing only `]`, and `[!]]` matches any single character except `]`. If there is no such closing `]`, the `[` is an ordinary literal character.
- A `-` that is the first or last character of a class, or that is escaped, is a literal `-`. An escaped character inside a class is a literal member of the class.
- A backslash escapes the next character: `\` followed by any character matches that character literally, so `\*` matches a literal `*` and `\a` matches `a`. This applies inside character classes as well. A backslash at the very end of the pattern matches a literal backslash.

import unittest

from globmatch import glob_match


class TestGlobMatch(unittest.TestCase):
    def test_literals_and_empty(self):
        self.assertTrue(glob_match("", ""))
        self.assertFalse(glob_match("", "a"))
        self.assertTrue(glob_match("abc", "abc"))
        self.assertFalse(glob_match("abc", "abd"))

    def test_question(self):
        self.assertTrue(glob_match("a?c", "abc"))
        self.assertFalse(glob_match("a?c", "ac"))

    def test_star(self):
        self.assertTrue(glob_match("*", ""))
        self.assertTrue(glob_match("a*c", "abbbc"))
        self.assertFalse(glob_match("a*c", "abbb"))
        self.assertTrue(glob_match("*.py", "x.py"))
        self.assertFalse(glob_match("*.py", "x.pyc"))

    def test_classes(self):
        self.assertTrue(glob_match("[abc]x", "bx"))
        self.assertFalse(glob_match("[abc]x", "dx"))
        self.assertTrue(glob_match("[a-z]9", "q9"))
        self.assertFalse(glob_match("[a-z]9", "Q9"))
        self.assertTrue(glob_match("[!a-z]9", "Q9"))

    def test_escape(self):
        self.assertTrue(glob_match(r"\*", "*"))
        self.assertFalse(glob_match(r"\*", "x"))
        self.assertTrue(glob_match(r"a\?b", "a?b"))

    def test_multiple_stars_small(self):
        self.assertTrue(glob_match("*a*a*b", "aaab"))
        self.assertFalse(glob_match("*a*a*b", "aaa"))


if __name__ == "__main__":
    unittest.main()
