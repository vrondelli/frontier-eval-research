"""Product catalog normalization."""

from decimal import Decimal, ROUND_HALF_UP
from fractions import Fraction
import math
from numbers import Integral


def _price_in_cents(value):
    """Return a non-negative integer, rounding positive halves upward."""
    if value is None or value < 0:
        return 0
    if isinstance(value, Integral):
        return int(value)
    if isinstance(value, Decimal):
        return int(value.quantize(Decimal("1"), rounding=ROUND_HALF_UP))
    if isinstance(value, Fraction):
        # Exact half-up rounding for rational values.
        return (2 * value.numerator + value.denominator) // (2 * value.denominator)
    return math.floor(value + 0.5)


def normalize_products(products):
    """Return normalized copies of the valid, first-seen products."""
    out = []
    seen_skus = set()

    for product in products:
        raw_sku = product.get("sku")
        raw_name = product.get("name")
        if raw_sku is None or raw_name is None:
            continue

        sku = str(raw_sku).strip()
        name = " ".join(str(raw_name).split())
        if not sku or not name:
            continue

        sku_key = sku.lower()
        if sku_key in seen_skus:
            continue
        seen_skus.add(sku_key)

        tags = []
        seen_tags = set()
        for tag in product.get("tags") or []:
            normalized_tag = str(tag).strip().lower()
            if normalized_tag and normalized_tag not in seen_tags:
                seen_tags.add(normalized_tag)
                tags.append(normalized_tag)

        out.append({
            "sku": sku,
            "name": name,
            "price_cents": _price_in_cents(product.get("price_cents")),
            "tags": tags,
        })

    return out
