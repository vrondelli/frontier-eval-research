product, with exactly the keys `sku`, `name`, `price_cents`, and `tags`.

Dimension 1: Products are deduplicated by sku: two products are duplicates when their sku values are equal after stripping surrounding whitespace and lowercasing, only the first occurrence is kept, and the relative order of kept products is preserved. Dimension 4: The tags in the output are a list of strings: each tag is stripped of surrounding whitespace and lowercased, empty tags are dropped, duplicate tags are removed keeping the first occurrence, and a missing or None tags value becomes an empty list.

"""Product catalog normalization."""

from decimal import Decimal, InvalidOperation, ROUND_HALF_UP


def normalize_products(products):
    """Return normalized copies of the products, leaving the input untouched."""
    out = []
    seen_skus = set()

    for product in products:
        raw_sku = product.get("sku")
        raw_name = product.get("name")

        if raw_sku is None or raw_name is None:
            continue

        sku = str(raw_sku).strip()
        name = " ".join(str(raw_name).split())
        if not sku or not name:
            continue

        sku_key = sku.lower()
        if sku_key in seen_skus:
            continue
        seen_skus.add(sku_key)

        raw_price = product.get("price_cents")
        if raw_price is None:
            price_cents = 0
        else:
            try:
                price = Decimal(str(raw_price))
                if price < 0:
                    price_cents = 0
                else:
                    price_cents = int(price.quantize(Decimal("1"), rounding=ROUND_HALF_UP))
            except (InvalidOperation, ValueError):
                # The contract specifies numeric prices; keep malformed values
                # from leaking into the normalized integer field.
                price_cents = 0

        raw_tags = product.get("tags")
        tags = []
        seen_tags = set()
        if raw_tags is not None:
            for raw_tag in raw_tags:
                tag = str(raw_tag).strip().lower()
                if tag and tag not in seen_tags:
                    tags.append(tag)
                    seen_tags.add(tag)

        out.append({
            "sku": sku,
            "name": name,
            "price_cents": price_cents,
            "tags": tags,
        })

    return out
