"""A small, durable, process-safe file-backed ledger."""

import fcntl
import json
import os
import tempfile
from contextlib import contextmanager


def _lock_path(path):
    return os.fspath(path) + ".lock"


@contextmanager
def _locked(path, exclusive):
    """Hold a stable sidecar lock while accessing/replacing the ledger."""
    lock_fd = os.open(_lock_path(path), os.O_CREAT | os.O_RDWR, 0o666)
    try:
        fcntl.flock(lock_fd, fcntl.LOCK_EX if exclusive else fcntl.LOCK_SH)
        yield
    finally:
        try:
            fcntl.flock(lock_fd, fcntl.LOCK_UN)
        finally:
            os.close(lock_fd)


def _read_records(path):
    try:
        with open(path, "r", encoding="utf-8") as store:
            records = json.load(store)
    except FileNotFoundError:
        return []
    if not isinstance(records, list):
        raise ValueError("invalid ledger store")
    return records


def _write_records(path, records):
    """Atomically replace the store and make the replacement durable."""
    path = os.fspath(path)
    parent = os.path.dirname(os.path.abspath(path))
    fd, temporary = tempfile.mkstemp(prefix=".ledger-", suffix=".tmp", dir=parent)
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as store:
            json.dump(records, store, ensure_ascii=True, separators=(",", ":"))
            store.flush()
            os.fsync(store.fileno())
        os.replace(temporary, path)
        # Persist the directory entry update as well as the file contents.
        dir_fd = os.open(parent, os.O_RDONLY | getattr(os, "O_DIRECTORY", 0))
        try:
            os.fsync(dir_fd)
        finally:
            os.close(dir_fd)
    finally:
        try:
            os.unlink(temporary)
        except FileNotFoundError:
            pass


def append_record(path, account, amount, note=""):
    if not isinstance(account, str) or not account:
        raise ValueError("account must be a non-empty string")
    if isinstance(amount, bool) or not isinstance(amount, int):
        raise ValueError("amount must be an int, not bool")
    if not isinstance(note, str):
        raise ValueError("note must be a string")

    with _locked(path, exclusive=True):
        records = _read_records(path)
        seq = len(records) + 1
        records.append(
            {"seq": seq, "account": account, "amount": amount, "note": note}
        )
        _write_records(path, records)
        return seq


def list_records(path):
    with _locked(path, exclusive=False):
        # Return fresh dictionaries with exactly the public record fields.
        records = _read_records(path)
        return [
            {
                "seq": record["seq"],
                "account": record["account"],
                "amount": record["amount"],
                "note": record["note"],
            }
            for record in records
        ]


def balance(path, account):
    with _locked(path, exclusive=False):
        records = _read_records(path)
        return sum(record["amount"] for record in records if record["account"] == account)


def count(path):
    with _locked(path, exclusive=False):
        return len(_read_records(path))
